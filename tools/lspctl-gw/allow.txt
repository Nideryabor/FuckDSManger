# /data/adb/lspctl-gw/allow.txt
# ============================================================================
#  lspctl 白名单 —— 逐字模板匹配（不是正则、不是前缀模糊匹配）
#
#  规则：
#    · 一行一条，'#' 开头是注释，空行忽略
#    · **不带空格结尾** = 只允许这条命令本身（不许带参数）
#    · **带空格结尾**   = 允许这条命令 + 任意合法参数
#
#  ⚠️ 铁律：只放【只读】或【可预期】的子命令。
#     任何能删模块 / 能改系统行为的命令，加之前先想三遍。
#
#  ⚠️ 改完不用重启网关（每次都重新读这个文件）
# ============================================================================

# ── 只读：状态查询 ──
lspctl status
lspctl --json status
lspctl -V
lspctl --version

# ── 只读：模块与作用域 ──
lspctl module list
lspctl --json module list
lspctl scope list
lspctl --json scope list

# ── 可写：作用域（这是我们要自动化的核心）──
# 允许带参数；参数只允许 [A-Za-z0-9._:/=-]
lspctl scope add 
lspctl scope remove 
lspctl scope set 

# ── 只读：hook 调试信息（诊断用）──
lspctl hook-debug 
lspctl --json hook-debug 

# ══════════════════════════════════════════════════════════════════════
#  ❌ 故意【不放】的高危命令（列在这提醒以后的自己）：
#
#  lspctl stop              ← 停 daemon = 所有模块失效
#  lspctl module enable/disable  ← 能关别人的模块
#  lspctl module remove     ← 能删模块配置
#
#  真要临时放开：**手改这个文件**（root:root 0700，尼尼改不了），
#  用完立刻删掉那行。别图省事常开。
